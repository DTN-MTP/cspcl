cmake_minimum_required(VERSION 3.10)
project(cspcl VERSION 1.0.0 LANGUAGES C)

# C standard
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Options
option(CSPCL_BUILD_TESTS "Build unit tests" ON)
option(CSPCL_USE_FREERTOS "Build for FreeRTOS (default: POSIX)" OFF)
option(CSPCL_DEBUG "Enable debug output" OFF)

# CSP library handling
# CSPCL requires CSP v1.6 (not v2.0 which has a different API)
# By default, use the included stubs for CSP v1.6 API
option(CSPCL_USE_SYSTEM_CSP "Use system-installed libcsp (must be v1.6)" ON)

# Path to the root of a *built* (not necessarily installed) libcsp repository.
# Set this to the top-level directory of the libcsp repo you cloned and built,
# e.g.:  cmake -DCSP_REPO_DIR=/home/user/libcsp ..
# The script expects the following layout (produced by a waf build):
#
#   <CSP_REPO_DIR>/
#     include/            <- source headers (csp/csp.h, ...)
#     build/
#       libcsp.a          <- static library  (waf default)
#       libcsp.so         <- shared library  (if built with --enable-shlib)
#       include/
#         csp/
#           csp_autoconfig.h  <- generated by waf (libcsp 1.6 default location)
#
# If you used a custom waf --out= directory, also set CSP_BUILD_DIR explicitly.
set(CSP_REPO_DIR  "" CACHE PATH
    "Root of the built libcsp repository (e.g. /home/user/libcsp)")
set(CSP_BUILD_DIR "" CACHE PATH
    "Override the build output directory inside the libcsp repo (default: <CSP_REPO_DIR>/build)")

# Required libcsp version
set(CSP_REQUIRED_VERSION "1.6")

set(CSP_FOUND FALSE)
set(USING_CSP_STUBS TRUE)

if(CSPCL_USE_SYSTEM_CSP)

    # ------------------------------------------------------------------
    # Validate CSP_REPO_DIR: it must be provided and must exist.
    # ------------------------------------------------------------------
    if(NOT CSP_REPO_DIR)
        message(FATAL_ERROR
            "CSPCL_USE_SYSTEM_CSP is ON but CSP_REPO_DIR is not set.\n"
            "Point it to your built libcsp repository:\n"
            "  cmake -DCSP_REPO_DIR=/path/to/libcsp ..\n"
            "Or disable system CSP to use the built-in stubs:\n"
            "  cmake -DCSPCL_USE_SYSTEM_CSP=OFF .."
        )
    endif()

    if(NOT IS_DIRECTORY "${CSP_REPO_DIR}")
        message(FATAL_ERROR
            "CSP_REPO_DIR does not exist or is not a directory: ${CSP_REPO_DIR}"
        )
    endif()

    # Resolve the build output directory
    if(CSP_BUILD_DIR)
        set(_CSP_BUILD "${CSP_BUILD_DIR}")
    else()
        set(_CSP_BUILD "${CSP_REPO_DIR}/build")
    endif()

    if(NOT IS_DIRECTORY "${_CSP_BUILD}")
        message(FATAL_ERROR
            "libcsp build directory not found: ${_CSP_BUILD}\n"
            "Make sure you have built libcsp first (e.g. 'python3 waf build' inside ${CSP_REPO_DIR}).\n"
            "If you used a custom waf --out= path, set -DCSP_BUILD_DIR=<path>."
        )
    endif()

    message(STATUS "libcsp repo dir:   ${CSP_REPO_DIR}")
    message(STATUS "libcsp build dir:  ${_CSP_BUILD}")

    # ------------------------------------------------------------------
    # Locate headers
    # libcsp 1.6 keeps public headers in <repo>/include/
    # waf may also write generated headers (csp_autoconfig.h) into
    # <build>/csp/ — include that directory too so nothing is missed.
    # ------------------------------------------------------------------
    find_path(CSP_INCLUDE_DIRS csp/csp.h
        HINTS
            "${CSP_REPO_DIR}/include"
        NO_DEFAULT_PATH          # do NOT fall back to system paths
    )

    # ------------------------------------------------------------------
    # Locate csp_autoconfig.h — generated by waf during the build.
    # waf can write it to different places depending on version:
    #   <build>/csp/csp_autoconfig.h          (most common, libcsp 1.6)
    #   <build>/include/csp/csp_autoconfig.h  (some waf configurations)
    # We need the *parent* of the csp/ subdirectory on the include path
    # so that #include <csp/csp_autoconfig.h> resolves correctly.
    # ------------------------------------------------------------------
    set(_CSP_GENERATED_INCLUDE "")

    # Check known waf output locations in priority order.
    # libcsp 1.6 waf writes csp_autoconfig.h to <build>/include/csp/ by default.
    foreach(_candidate
        "${_CSP_BUILD}/include"      # <repo>/build/include/csp/  <- libcsp 1.6 default
        "${_CSP_BUILD}"              # <repo>/build/csp/
        "${CSP_REPO_DIR}/include"    # <repo>/include/csp/  (in-source builds)
    )
        if(EXISTS "${_candidate}/csp/csp_autoconfig.h")
            set(_CSP_GENERATED_INCLUDE "${_candidate}")
            message(STATUS "Found csp_autoconfig.h in: ${_candidate}/csp/")
            break()
        endif()
    endforeach()

    if(NOT _CSP_GENERATED_INCLUDE)
        # Last resort: recursive search across the whole repo tree
        file(GLOB_RECURSE _CSP_AUTOCONFIG_FILE "${CSP_REPO_DIR}/csp_autoconfig.h")
        if(_CSP_AUTOCONFIG_FILE)
            list(GET _CSP_AUTOCONFIG_FILE 0 _CSP_AUTOCONFIG_FILE)
            # Strip /csp/csp_autoconfig.h to get the include root
            get_filename_component(_CSP_AUTOCONFIG_CSP_DIR "${_CSP_AUTOCONFIG_FILE}" DIRECTORY)
            get_filename_component(_CSP_GENERATED_INCLUDE  "${_CSP_AUTOCONFIG_CSP_DIR}" DIRECTORY)
            message(STATUS "Found csp_autoconfig.h (recursive) under: ${_CSP_GENERATED_INCLUDE}")
        else()
            message(WARNING
                "csp_autoconfig.h not found anywhere under ${CSP_REPO_DIR}. "
                "Make sure libcsp has been fully built (python3 waf build) before running cmake. "
                "If the build directory is non-standard, set -DCSP_BUILD_DIR=<path>."
            )
        endif()
    endif()

    # ------------------------------------------------------------------
    # Locate the library
    # waf places libcsp.a (and optionally libcsp.so) directly in <build>/
    # ------------------------------------------------------------------
    find_library(CSP_LIBRARIES csp
        HINTS
            "${_CSP_BUILD}"
        NO_DEFAULT_PATH          # do NOT pick up a system libcsp by accident
    )

    # ------------------------------------------------------------------
    # Version check — parse CSP_VERSION_MAJOR / CSP_VERSION_MINOR from
    # the repo headers.  libcsp 1.6 defines these in csp/csp.h itself;
    # some builds put them in csp/csp_types.h or csp/csp_version.h.
    # ------------------------------------------------------------------
    if(CSP_INCLUDE_DIRS)
        set(_CSP_DETECTED_VERSION "")

        # Try dedicated version header first, then fall back to csp.h
        foreach(_hdr
            "${CSP_INCLUDE_DIRS}/csp/csp_version.h"
            "${CSP_INCLUDE_DIRS}/csp/csp_types.h"
            "${CSP_INCLUDE_DIRS}/csp/csp.h"
        )
            if(EXISTS "${_hdr}" AND NOT _CSP_DETECTED_VERSION)
                file(STRINGS "${_hdr}" _CSP_VER_MAJOR_LINE
                     REGEX "#define[ \t]+CSP_VERSION_MAJOR[ \t]+[0-9]+")
                file(STRINGS "${_hdr}" _CSP_VER_MINOR_LINE
                     REGEX "#define[ \t]+CSP_VERSION_MINOR[ \t]+[0-9]+")
                if(_CSP_VER_MAJOR_LINE AND _CSP_VER_MINOR_LINE)
                    string(REGEX REPLACE ".*CSP_VERSION_MAJOR[ \t]+([0-9]+).*" "\\1"
                           _CSP_MAJOR "${_CSP_VER_MAJOR_LINE}")
                    string(REGEX REPLACE ".*CSP_VERSION_MINOR[ \t]+([0-9]+).*" "\\1"
                           _CSP_MINOR "${_CSP_VER_MINOR_LINE}")
                    set(_CSP_DETECTED_VERSION "${_CSP_MAJOR}.${_CSP_MINOR}")
                endif()
            endif()
        endforeach()

        if(_CSP_DETECTED_VERSION)
            message(STATUS "Detected libcsp version: ${_CSP_DETECTED_VERSION}")

            # Major version must match (v2 has an incompatible API)
            string(REGEX MATCH "^([0-9]+)" _CSP_DETECTED_MAJOR "${_CSP_DETECTED_VERSION}")
            string(REGEX MATCH "^([0-9]+)" _CSP_REQ_MAJOR      "${CSP_REQUIRED_VERSION}")

            if(NOT _CSP_DETECTED_MAJOR EQUAL _CSP_REQ_MAJOR)
                message(FATAL_ERROR
                    "libcsp at ${CSP_REPO_DIR} is version ${_CSP_DETECTED_VERSION}, "
                    "but this project requires v${CSP_REQUIRED_VERSION}.x — "
                    "v2.x has an incompatible API.\n"
                    "Point CSP_REPO_DIR to a v${CSP_REQUIRED_VERSION} repository."
                )
            endif()

            # Within the same major, require at least the specified minor
            if(_CSP_DETECTED_VERSION VERSION_LESS CSP_REQUIRED_VERSION)
                message(FATAL_ERROR
                    "libcsp at ${CSP_REPO_DIR} is version ${_CSP_DETECTED_VERSION}, "
                    "but >= ${CSP_REQUIRED_VERSION} is required."
                )
            endif()
        else()
            message(WARNING
                "Could not detect libcsp version from headers in ${CSP_INCLUDE_DIRS}. "
                "Proceeding, but verify the repository is v${CSP_REQUIRED_VERSION}."
            )
        endif()
    endif()

    # ------------------------------------------------------------------
    # Decide whether we have everything we need
    # ------------------------------------------------------------------
    if(CSP_INCLUDE_DIRS AND CSP_LIBRARIES)
        set(CSP_FOUND TRUE)
        set(USING_CSP_STUBS FALSE)
        # Combine source headers + generated headers into one list.
        # Use a normal (non-cache) variable so we can freely append to it.
        set(CSP_ALL_INCLUDE_DIRS "${CSP_INCLUDE_DIRS}")
        if(_CSP_GENERATED_INCLUDE)
            list(APPEND CSP_ALL_INCLUDE_DIRS "${_CSP_GENERATED_INCLUDE}")
        endif()
        list(REMOVE_DUPLICATES CSP_ALL_INCLUDE_DIRS)
    else()
        if(NOT CSP_INCLUDE_DIRS)
            message(FATAL_ERROR
                "csp/csp.h not found under ${CSP_REPO_DIR}/include.\n"
                "Check that CSP_REPO_DIR points to the correct libcsp repository."
            )
        endif()
        if(NOT CSP_LIBRARIES)
            message(FATAL_ERROR
                "libcsp library not found under ${_CSP_BUILD}.\n"
                "Build libcsp first: cd ${CSP_REPO_DIR} && python3 waf build\n"
                "If you used --out=<dir>, pass -DCSP_BUILD_DIR=<dir> to cmake."
            )
        endif()
    endif()

else()
    message(STATUS "Using CSP v1.6 stubs (pass -DCSPCL_USE_SYSTEM_CSP=ON -DCSP_REPO_DIR=<path> to use a real libcsp build)")
    set(CSP_INCLUDE_DIRS ${CMAKE_SOURCE_DIR}/stubs)
    set(CSP_ALL_INCLUDE_DIRS ${CMAKE_SOURCE_DIR}/stubs)
    set(CSP_LIBRARIES "")
    set(USING_CSP_STUBS TRUE)
endif()

# CSPCL library
set(CSPCL_SOURCES src/cspcl.c)

# Add stub implementation when not using system CSP
if(USING_CSP_STUBS)
    list(APPEND CSPCL_SOURCES stubs/csp_stub.c)
endif()

add_library(cspcl ${CSPCL_SOURCES})

target_include_directories(cspcl
    PUBLIC
        $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/src>
        $<INSTALL_INTERFACE:include>
    PRIVATE
        ${CSP_ALL_INCLUDE_DIRS}
)

target_link_libraries(cspcl
    PRIVATE
        ${CSP_LIBRARIES}
)

# Platform-specific settings
if(CSPCL_USE_FREERTOS)
    target_compile_definitions(cspcl PRIVATE FREERTOS)
else()
    target_compile_definitions(cspcl PRIVATE __linux__)
    target_link_libraries(cspcl PRIVATE rt)
endif()

# Debug settings
if(CSPCL_DEBUG)
    target_compile_definitions(cspcl PRIVATE CSPCL_DEBUG CSPCL_DEBUG_VERBOSE)
endif()

# Compiler warnings
target_compile_options(cspcl PRIVATE
    -Wall -Wextra -Wpedantic
    $<$<CONFIG:Debug>:-g -O0>
    $<$<CONFIG:Release>:-O2>
)

# Tests
if(CSPCL_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()

# Installation
install(TARGETS cspcl
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
)

install(FILES src/cspcl.h src/cspcl_config.h
    DESTINATION include
)

# Package configuration
include(CMakePackageConfigHelpers)
write_basic_package_version_file(
    "${CMAKE_BINARY_DIR}/cspcl-config-version.cmake"
    VERSION ${PROJECT_VERSION}
    COMPATIBILITY SameMajorVersion
)

# Summary
message(STATUS "")
message(STATUS "CSPCL Configuration:")
message(STATUS "  Version:         ${PROJECT_VERSION}")
message(STATUS "  Required CSP:    ${CSP_REQUIRED_VERSION}.x")
message(STATUS "  CSP repo dir:    ${CSP_REPO_DIR}")
message(STATUS "  CSP build dir:   ${_CSP_BUILD}")
message(STATUS "  Using CSP stubs: ${USING_CSP_STUBS}")
message(STATUS "  CSP includes:    ${CSP_ALL_INCLUDE_DIRS}")
message(STATUS "  CSP libraries:   ${CSP_LIBRARIES}")
message(STATUS "  Build tests:     ${CSPCL_BUILD_TESTS}")
message(STATUS "  Build examples:  ${CSPCL_BUILD_EXAMPLES}")
message(STATUS "  FreeRTOS:        ${CSPCL_USE_FREERTOS}")
message(STATUS "  Debug:           ${CSPCL_DEBUG}")
message(STATUS "")
